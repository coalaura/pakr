cmake_minimum_required(VERSION 3.20)
project(pakr_wim C)

include(CheckIncludeFile)
include(CheckFunctionExists)
include(CheckCSourceCompiles)

add_compile_definitions(_GNU_SOURCE _LARGEFILE_SOURCE _FILE_OFFSET_BITS=64 BUILDING_WIMLIB HAVE_CONFIG_H PACKAGE_VERSION="1.14.5" PACKAGE_BUGREPORT="https://wimlib.net/forums/")
set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE -D_FILE_OFFSET_BITS=64)
foreach(header alloca.h byteswap.h dlfcn.h endian.h errno.h glob.h inttypes.h machine/endian.h stdarg.h stddef.h stdint.h stdio.h stdlib.h strings.h string.h sys/byteorder.h sys/endian.h sys/file.h sys/stat.h sys/syscall.h sys/sysctl.h sys/times.h sys/types.h sys/xattr.h time.h unistd.h utime.h)
	string(TOUPPER "HAVE_${header}" variable)
	string(REGEX REPLACE "[/.]" "_" variable "${variable}")
	check_include_file("${header}" "${variable}")
endforeach()
foreach(function fdopendir flock fsetxattr fstatat futimens getopt_long_only lgetxattr llistxattr lsetxattr mempcpy openat posix_fallocate readlinkat utimensat)
	string(TOUPPER "HAVE_${function}" variable)
	check_function_exists("${function}" "${variable}")
endforeach()
check_c_source_compiles("#include <sys/stat.h>\nint main(void) { struct stat st; return st.st_mtim.tv_nsec; }" HAVE_STAT_NANOSECOND_PRECISION)
set(HAVE_PTHREAD 1)
set(STDC_HEADERS 1)
file(READ "${SOURCE}/config.h.in" configuration)
string(REGEX REPLACE "#undef ([A-Z0-9_]+)" "#cmakedefine \\1 1" configuration "${configuration}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h.in" "${configuration}")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/config.h.in" "${CMAKE_CURRENT_BINARY_DIR}/config.h")

file(GLOB sources "${SOURCE}/src/*.c")
list(FILTER sources EXCLUDE REGEX "/(test_support|ntfs-3g_[^/]+)\\.c$")
if(WIN32)
	list(FILTER sources EXCLUDE REGEX "/unix_[^/]+\\.c$")
	add_compile_definitions(_POSIX _POSIX_THREAD_SAFE_FUNCTIONS UNICODE _UNICODE _CRT_NON_CONFORMING_SWPRINTFS __MINGW_USE_VC2005_COMPAT)
	add_compile_options(-mno-ms-bitfields)
else()
	list(FILTER sources EXCLUDE REGEX "/(win32_[^/]+|wimboot)\\.c$")
endif()
add_library(wim STATIC ${sources} "${CMAKE_CURRENT_SOURCE_DIR}/../../internal/native/wim/wim.c")
set_target_properties(wim PROPERTIES C_STANDARD 99 C_EXTENSIONS ON)
target_compile_options(wim PRIVATE -fno-common)
target_include_directories(wim PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${SOURCE}/include")
install(TARGETS wim ARCHIVE DESTINATION lib)
install(FILES "${SOURCE}/include/wimlib.h" DESTINATION include)
