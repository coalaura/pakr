cmake_minimum_required(VERSION 3.16)
project(pakr_kanzi LANGUAGES C CXX)
set(KANZI_ENABLE_NATIVE_OPTIMIZATIONS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCE}" upstream EXCLUDE_FROM_ALL)
add_library(pakr_kanzi STATIC ../../internal/native/kanzi/kanzi.cc)
target_compile_features(pakr_kanzi PRIVATE cxx_std_17)
target_compile_options(pakr_kanzi PRIVATE -fno-rtti)
target_include_directories(pakr_kanzi PRIVATE "${SOURCE}/src")
target_link_libraries(pakr_kanzi PRIVATE libkanzi)
install(TARGETS pakr_kanzi libkanzi ARCHIVE DESTINATION lib)
